        var elements = document.querySelectorAll('[id^="nv-waveform-container"]');

        elements.forEach(function(element) {
            var match = element.id.match(/\d+/);
            if (!match) return;
            var index = parseInt(match[0], 10);
            var audioSrc = nvAudioSources[index - 1];
            var jsonSrc = nvJsonSources[index - 1];
            if (!audioSrc || !jsonSrc) return;
            buildPlayer(element, index, audioSrc, jsonSrc);

        function stopAllAudio() {
            var playingAudios = document.querySelectorAll('.nv-listen audio.nv-playing');
            playingAudios.forEach(function(a) {
                a.pause();
                a.classList.remove('nv-playing');
                a.currentTime = 0;
            });
            document.querySelectorAll('.nv-listen .fa-pause').forEach(function(icon) {
                icon.classList.replace('fa-pause', 'fa-play');
            });
        }

        function buildPlayer(container, index, audioSrc, jsonSrc) {
            var clickProgress = 0;

            var div = document.createElement('div');
            div.setAttribute('class', 'waveform');

            var canvas = document.createElement('canvas');
            canvas.width = 1000;
            canvas.height = 60;
            canvas.setAttribute('class', 'waveform-canvas');
            canvas.setAttribute('data-audio-src', audioSrc);
            canvas.setAttribute('data-json-src', jsonSrc);

            var audio = document.createElement('audio');
            audio.src = audioSrc;
            audio.preload = 'auto';
            audio.controls = true;

            var playButton = document.createElement('div');
            playButton.setAttribute('id', 'nv-btn-contain' + index);
            playButton.setAttribute('class', 'demo-btn-contain demo-btn-play');
            playButton.setAttribute('data-audio-src', audioSrc);
            playButton.innerHTML = '<i id="nv-pbtn' + index + '" class="demo-btn fas fa-play fa-inverse"></i>';

            playButton.addEventListener('click', function() {
                var icon = playButton.querySelector('i');
                if (icon.classList.contains('fa-play')) {
                    stopAllAudio();
                    icon.classList.remove('fa-play');
                    icon.classList.add('fa-pause');
                    audio.play();
                    audio.classList.add('nv-playing');
                } else {
                    icon.classList.remove('fa-pause');
                    icon.classList.add('fa-play');
                    audio.classList.remove('nv-playing');
                    audio.pause();
                }
            });

            var waveform = document.createElement('div');
            waveform.setAttribute('id', 'nv-waveform' + index);
            waveform.setAttribute('class', 'waveform-contain');
            waveform.appendChild(canvas);
            waveform.appendChild(audio);

            div.appendChild(playButton);
            div.appendChild(waveform);
            container.appendChild(div);

            fetch(jsonSrc)
                .then(function(response) { return response.json(); })
                .then(function(data) {
                    var ctx = canvas.getContext('2d');
                    var samples = data.samples;
                    drawWaveform(samples, ctx, canvas.width, canvas.height, 0);

                    canvas.addEventListener('click', function(e) {
                        stopAllAudio();
                        var rect = canvas.getBoundingClientRect();
                        var clickX = e.clientX - rect.left;
                        clickProgress = clickX / rect.width;
                        audio.currentTime = clickProgress * audio.duration;

                        var icon = playButton.querySelector('i');
                        if (icon.classList.contains('fa-play')) {
                            icon.classList.remove('fa-play');
                            icon.classList.add('fa-pause');
                        }
                        audio.play();
                        audio.classList.add('nv-playing');
                    });

                    audio.addEventListener('loadedmetadata', function() {
                        audio.currentTime = clickProgress * audio.duration;
                    });

                    audio.addEventListener('timeupdate', function() {
                        var progress = audio.currentTime / audio.duration;
                        drawWaveform(samples, ctx, canvas.width, canvas.height, progress);
                    });

                    audio.addEventListener('ended', function() {
                        stopAllAudio();
                    });
                })
                .catch(function(error) { console.warn('[nv-listen] JSON fetch failed for index ' + index + ':', error); });
        }

        function drawWaveform(samples, ctx, width, height, progress) {
            var maxAbsoluteValue = Math.max.apply(null, samples.map(Math.abs));
            var halfHeight = height / 2;
            var step = halfHeight / maxAbsoluteValue;
            var endIndex = Math.floor(samples.length * progress);

            ctx.clearRect(0, 0, width, height);

            for (var i = 0; i < samples.length; i++) {
                var normalizedSample = Math.abs(samples[i]) * step;
                var x = (i / samples.length) * width;
                var y1 = halfHeight - normalizedSample;
                var y2 = halfHeight + normalizedSample;

                if (i <= endIndex) {
                    ctx.strokeStyle = '#ffc767';
                } else {
                    ctx.strokeStyle = 'rgba(255, 255, 255, 0.28)';
                }

                ctx.lineWidth = 2;
                ctx.beginPath();
                ctx.moveTo(x, y1);
                ctx.lineTo(x, y2);
                ctx.stroke();
            }
        }
    });
})();
